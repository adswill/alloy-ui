// A starter Alloy app: a counter. Run it with `alloy dev`, package it with
// `alloy bundle`.

use alloy_ui::prelude::*;

fn main() {
    sketch(0i64, |n| {
        col![
            col![heading("@TITLE@"), text(format!("{n} so far")).dim()].gap(2.0),
            row![
                button("Subtract").act(|n| *n -= 1),
                button("Add").primary().act(|n| *n += 1),
            ]
            .align_center(),
        ]
        .gap(18.0)
        .pad(24.0)
        .fill()
    })
    .run_sized("@TITLE@", 360.0, 200.0)
}
